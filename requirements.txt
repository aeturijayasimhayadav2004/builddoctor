# BuildDoctor - Phase 2 dependencies

# Web framework + server (Phase 0)
fastapi>=0.110
uvicorn>=0.27

# Async HTTP client for calling the GitHub API (Phase 1)
httpx>=0.27

# Loads secrets from .env (Phase 1)
python-dotenv>=1.0

# Grok is OpenAI-compatible, so the openai SDK is pointed at
# https://api.x.ai/v1 rather than at OpenAI (Phase 2)
openai>=1.40

# Postgres storage: ORM/engine plus the driver it delegates to.
# The URL scheme postgresql+psycopg:// selects psycopg 3 specifically.
# [binary] ships prebuilt wheels, so no C compiler is needed. (Phase 3)
sqlalchemy>=2.0
psycopg[binary]>=3.1

# The classify -> route -> act state machine in graph.py (Phase 4)
langgraph>=0.2

# MCP server (mcp_server.py) and the client graph.py uses to reach it.
# Note 2.x renamed FastMCP to MCPServer; 1.x code will not run on this.
mcp>=2.1

# Local embedding model for BuildDoctor's memory (Phase 6).
# Runs on CPU, weights baked into the image at build time - see Dockerfile.
# NOTE: this pulls in PyTorch. The Dockerfile installs the CPU-only build
# first, on purpose; plain `pip install torch` fetches the CUDA stack.
sentence-transformers>=3.0

# Registers the pgvector column type with SQLAlchemy so `embedding` maps to
# a Python list instead of an opaque string. (Phase 6)
pgvector>=0.2

# GitHub App auth: sign the RS256 JWT that is traded for an installation
# access token (Phase 11). The [crypto] extra pulls in `cryptography`, which
# is what actually does the RSA signing - PyJWT alone cannot do RS256.
pyjwt[crypto]>=2.8

# Signs the dashboard's session cookie (Phase 12). Starlette's
# SessionMiddleware imports this lazily and raises AssertionError at startup
# without it, so it is pinned explicitly rather than left to arrive as
# somebody else's transitive dependency.
itsdangerous>=2.1
